cmake_minimum_required(VERSION 3.25)
project(GameBoundaryGuardFixture LANGUAGES NONE)
include("${HLCLIENT_ROOT}/cmake/GameModuleBoundary.cmake")
add_library(hlclient_fixture_core INTERFACE)
add_library(hlclient_fixture_bridge INTERFACE)
add_library(hlclient_fixture_game INTERFACE)
add_library(hlclient INTERFACE)
set_property(TARGET hlclient PROPERTY HLCLIENT_GAME_COMPOSITION TRUE)
target_link_libraries(hlclient INTERFACE hlclient_fixture_game)
set_property(TARGET hlclient_fixture_game PROPERTY HLCLIENT_GAME_MODULE TRUE)
add_library(game_alias ALIAS hlclient_fixture_game)
add_library(hlclient::fixture_game ALIAS hlclient_fixture_game)
add_library(hlclient::fixture_core ALIAS hlclient_fixture_core)
add_library(hlclient::fixture_bridge ALIAS hlclient_fixture_bridge)
if(CASE STREQUAL "direct")
    target_link_libraries(hlclient_fixture_core INTERFACE hlclient_fixture_game)
elseif(CASE STREQUAL "alias")
    target_link_libraries(hlclient_fixture_core INTERFACE hlclient::fixture_game)
elseif(CASE STREQUAL "transitive")
    target_link_libraries(hlclient_fixture_core INTERFACE hlclient_fixture_bridge)
    target_link_libraries(hlclient_fixture_bridge INTERFACE game_alias)
elseif(CASE STREQUAL "generator")
    target_link_libraries(hlclient_fixture_core INTERFACE "$<LINK_ONLY:hlclient::fixture_game>")
elseif(CASE STREQUAL "source")
    target_sources(hlclient_fixture_core INTERFACE src/games/halflife/forbidden.hpp)
elseif(CASE STREQUAL "object")
    # Guard evaluation runs before generation checks the object-library kind.
    # This proves TARGET_OBJECTS cannot smuggle a concrete target past links.
    target_sources(hlclient_fixture_core INTERFACE "$<TARGET_OBJECTS:game_alias>")
elseif(CASE STREQUAL "include")
    target_sources(hlclient_fixture_core INTERFACE bridge.hpp)
elseif(CASE STREQUAL "allowed")
    target_link_libraries(hlclient_fixture_game INTERFACE hlclient::fixture_core)
    target_link_libraries(hlclient_fixture_core INTERFACE hlclient::fixture_bridge)
    # Literal include path components must not be interpreted as targets.
    target_sources(hlclient_fixture_core INTERFACE include/hlclient/safe.hpp)
else()
    message(FATAL_ERROR "Unknown guard case")
endif()
hlclient_check_game_module_boundaries()
